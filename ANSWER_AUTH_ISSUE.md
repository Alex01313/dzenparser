```python
def check_if_auth_required(self, response):
    """Check if the article requires authentication"""
    # Проверка редиректа на SSO
    if 'sso.passport.yandex.ru' in response.url:
        return True, "Redirected to Yandex authentication page"
    
    # Проверка размера страницы
    if len(response.content) < 5000:
        return True, "Page too small - likely requires auth"
    
    # Проверка на скрипты авторизации
    if b'form.submit()' in response.content.lower():
        return True, "Page contains auth redirect"
    
    return False, "OK"
```
